# Sessions command implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `sessions` command that lists recent Claude Code, OpenCode, and agy conversations grouped by git repo, and a `sessions tui` fzf picker that can copy or resume one.

**Architecture:** Three adapters read the CLIs' own files into one `Session` record. A grouper drops anything outside the window and rolls directories up to the git toplevel. A printer and an fzf runner are the only two consumers. The program stores nothing.

**Tech Stack:** Python 3.11+ standard library, setuptools console script, pytest, fzf 0.60+ (`--accept-nth`), `pbcopy`.

**Spec:** `docs/superpowers/specs/2026-11-19-sessions-design.md`

This plan is the original build. The picker has since grown: an empty query stays on the recent window, and any query searches prompts and replies across every session. The spec describes that behavior.

## Global Constraints

- Runtime imports are the Python standard library. pytest and setuptools are install and test tools only.
- Each run reads Claude, OpenCode, and agy and keeps nothing of its own.
- `--since` defaults to `7d` and accepts a positive integer plus `h` or `d`.
- The repo key is the git toplevel when `git -C <directory> rev-parse --show-toplevel` succeeds, otherwise the recorded directory.
- Repos are ordered by newest session, then path. Sessions inside a repo are ordered newest first, then tool name, then id.
- The resume line uses the recorded directory: `cd <quoted directory> && <tool command>`.
- Claude resumes with `claude --resume <id>`. OpenCode resumes with `opencode --session <id>`. agy resumes with `agy --conversation <id>`.
- A title is one line of at most 100 characters with internal whitespace collapsed. An empty title or a title that starts with `/` is discarded, then the next source is used, then `untitled`.
- One adapter failure prints one stderr warning and still shows the other tools, exit 0. All three failing exits 1 without the empty-window line. A bad `--since` exits 1. An empty window prints `no sessions in the last <token>` and exits 0.
- `sessions tui` is fzf. Enter runs the pasteable line. `ctrl-y` copies it and leaves fzf open. Missing fzf prints the list and `fzf: command not found` on stderr, exit 0. `pbcopy` failure prints the line on stderr.
- Warnings go to stderr. The list goes to stdout.
- Tests use fixtures only. They do not read the real stores and do not launch Claude, OpenCode, agy, or fzf.

```python
import re
import shlex
from dataclasses import dataclass
from datetime import datetime, timedelta
from pathlib import Path

_SINCE = re.compile(r"(\d+)([hd])")


@dataclass(frozen=True)
class Session:
    tool: str
    id: str
    directory: Path
    title: str
    updated: datetime


class AdapterError(Exception):
    def __init__(self, tool: str, message: str) -> None:
        super().__init__(message)
        self.tool = tool
        self.message = message


def parse_since(value: str) -> timedelta:
    match = _SINCE.fullmatch(value)
    if match is None:
        raise ValueError(value)
    amount = int(match.group(1))
    if amount < 1:
        raise ValueError(value)
    if match.group(2) == "h":
        return timedelta(hours=amount)
    return timedelta(days=amount)


def one_line(text: str) -> str:
    stripped = text.strip()
    if not stripped:
        return ""
    line = " ".join(stripped.splitlines()[0].split())
    if not line or line.startswith("/"):
        return ""
    return line[:100]


def resume_line(session: Session) -> str:
    commands = {
        "claude": "claude --resume",
        "opencode": "opencode --session",
        "agy": "agy --conversation",
    }
    command = f"{commands[session.tool]} {shlex.quote(session.id)}"
    return f"cd {shlex.quote(str(session.directory))} && {command}"
```

`sessions/group.py`:

```python
import subprocess
from collections.abc import Callable
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path

from sessions.model import Session


@dataclass(frozen=True)
class RepoGroup:
    repo: Path
    sessions: tuple[Session, ...]


def git_toplevel(directory: Path) -> Path:
    try:
        output = subprocess.check_output(
            ["git", "-C", str(directory), "rev-parse", "--show-toplevel"],
            stderr=subprocess.DEVNULL,
            text=True,
        )
    except (subprocess.CalledProcessError, FileNotFoundError, OSError):
        return directory
    return Path(output.strip())


def group_sessions(
    sessions: list[Session],
    *,
    cutoff: datetime,
    toplevel: Callable[[Path], Path] = git_toplevel,
) -> list[RepoGroup]:
    kept = [session for session in sessions if session.updated >= cutoff]
    cache: dict[Path, Path] = {}

    def repo_of(session: Session) -> Path:
        if session.directory not in cache:
            cache[session.directory] = toplevel(session.directory)
        return cache[session.directory]

    buckets: dict[Path, list[Session]] = {}
    for session in kept:
        buckets.setdefault(repo_of(session), []).append(session)

    groups = []
    for repo, items in buckets.items():
        ordered = tuple(sorted(items, key=lambda item: (-item.updated.timestamp(), item.tool, item.id)))
        groups.append(RepoGroup(repo, ordered))
    groups.sort(key=lambda group: (-group.sessions[0].updated.timestamp(), str(group.repo)))
    return groups
```

```python
import json
from datetime import datetime, timezone
from pathlib import Path

from sessions.model import AdapterError, Session, one_line


def load_claude(root: Path) -> list[Session]:
    projects = root / "projects"
    if not projects.is_dir():
        raise AdapterError("claude", "session store not found")
    history = _history_titles(root / "history.jsonl")
    sessions = []
    for path in sorted(projects.rglob("*.jsonl")):
        session = _session_from_transcript(path, history)
        if session is not None:
            sessions.append(session)
    return sessions


def _history_titles(path: Path) -> dict[str, str]:
    titles: dict[str, str] = {}
    if not path.is_file():
        return titles
    for obj in _objects(path):
        session_id = obj.get("sessionId")
        if not isinstance(session_id, str) or session_id in titles:
            continue
        title = one_line(str(obj.get("display") or ""))
        if title:
            titles[session_id] = title
    return titles


def _session_from_transcript(path: Path, history: dict[str, str]) -> Session | None:
    file_id = path.stem
    cwd = ""
    updated: datetime | None = None
    custom = ""
    user_title = ""
    for obj in _objects(path):
        if obj.get("isSidechain") is True:
            continue
        session_id = obj.get("sessionId")
        if isinstance(session_id, str) and session_id != file_id:
            continue
        if not cwd and isinstance(obj.get("cwd"), str):
            cwd = obj["cwd"]
        parsed = _parse_time(obj.get("timestamp"))
        if parsed is not None and (updated is None or parsed > updated):
            updated = parsed
        if obj.get("type") == "custom-title":
            title = one_line(str(obj.get("customTitle") or ""))
            if title:
                custom = title
        if obj.get("type") == "user" and not user_title:
            user_title = one_line(_message_text(obj))
    if not cwd or updated is None:
        return None
    title = custom or history.get(file_id) or user_title or "untitled"
    return Session("claude", file_id, Path(cwd), title[:100], updated)


def _message_text(obj: dict) -> str:
    message = obj.get("message")
    if not isinstance(message, dict):
        return ""
    content = message.get("content")
    if isinstance(content, str):
        return content
    if isinstance(content, list):
        chunks = []
        for part in content:
            if isinstance(part, dict) and part.get("type") == "text" and isinstance(part.get("text"), str):
                chunks.append(part["text"])
        return "\n".join(chunks)
    return ""


def _parse_time(value: object) -> datetime | None:
    if not isinstance(value, str):
        return None
    try:
        return datetime.fromisoformat(value.replace("Z", "+00:00")).astimezone(timezone.utc)
    except ValueError:
        return None


def _objects(path: Path) -> list[dict]:
    objects = []
    try:
        lines = path.read_text(encoding="utf-8").splitlines()
    except OSError:
        return objects
    for line in lines:
        line = line.strip()
        if not line:
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            continue
        if isinstance(obj, dict):
            objects.append(obj)
    return objects
```

```python
import sqlite3
from datetime import datetime, timezone
from pathlib import Path

from sessions.model import AdapterError, Session, one_line


def load_opencode(db_path: Path) -> list[Session]:
    if not db_path.is_file():
        raise AdapterError("opencode", "session store not found")
    try:
        connection = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    except sqlite3.OperationalError as exc:
        raise _db_error(exc) from exc
    try:
        try:
            rows = connection.execute(
                """
                SELECT id, title, directory, time_updated
                FROM session_v2
                WHERE time_archived IS NULL
                """
            ).fetchall()
        except sqlite3.OperationalError as exc:
            raise _db_error(exc) from exc
    finally:
        connection.close()
    sessions = []
    for session_id, title, directory, updated_ms in rows:
        text = one_line(title or "") or "untitled"
        updated = datetime.fromtimestamp(updated_ms / 1000, timezone.utc)
        sessions.append(Session("opencode", session_id, Path(directory), text, updated))
    return sessions


def _db_error(exc: sqlite3.OperationalError) -> AdapterError:
    if "locked" in str(exc).lower():
        return AdapterError("opencode", "database is locked")
    return AdapterError("opencode", "unreadable database")
```

```python
import json
import sqlite3
import urllib.parse
from datetime import datetime, timezone
from pathlib import Path

from sessions.model import AdapterError, Session, one_line


def load_agy(db_path: Path) -> list[Session]:
    if not db_path.is_file():
        raise AdapterError("agy", "session store not found")
    try:
        connection = sqlite3.connect(f"file:{db_path}?mode=ro", uri=True)
    except sqlite3.OperationalError as exc:
        raise AdapterError("agy", "unreadable database") from exc
    try:
        try:
            rows = connection.execute(
                """
                SELECT conversation_id, title, last_modified_time, workspace_uris
                FROM conversation_summaries
                WHERE nesting_depth = 0
                """
            ).fetchall()
        except sqlite3.OperationalError as exc:
            raise AdapterError("agy", "unreadable database") from exc
    finally:
        connection.close()
    sessions = []
    for conversation_id, title, modified, workspace_uris in rows:
        directory = _first_directory(workspace_uris)
        if directory is None:
            continue
        text = one_line(title or "") or "untitled"
        updated = datetime.fromisoformat(modified).astimezone(timezone.utc)
        sessions.append(Session("agy", conversation_id, directory, text, updated))
    return sessions


def _first_directory(workspace_uris: str) -> Path | None:
    try:
        values = json.loads(workspace_uris)
    except json.JSONDecodeError:
        return None
    if not isinstance(values, list):
        return None
    for value in values:
        if not isinstance(value, str) or not value.startswith("file://"):
            continue
        parsed = urllib.parse.urlparse(value)
        if parsed.path:
            return Path(urllib.parse.unquote(parsed.path))
    return None
```

```python
from datetime import datetime, tzinfo

from sessions.group import RepoGroup
from sessions.model import resume_line


def age_label(then: datetime, now: datetime) -> str:
    seconds = int((now - then).total_seconds())
    if seconds < 60:
        return "just now"
    if seconds < 3600:
        return f"{seconds // 60}m ago"
    if seconds < 86400:
        return f"{seconds // 3600}h ago"
    return f"{seconds // 86400}d ago"


def render(
    groups: list[RepoGroup],
    *,
    now: datetime,
    since_label: str,
    zone: tzinfo | None = None,
) -> str:
    if not groups:
        return f"no sessions in the last {since_label}\n"
    if zone is None:
        zone = datetime.now().astimezone().tzinfo
    blocks = []
    for group in groups:
        lines = [f"{group.repo}  touched {age_label(group.sessions[0].updated, now)}"]
        for session in group.sessions:
            stamp = session.updated.astimezone(zone).strftime("%m-%d %H:%M")
            lines.append(f"  {session.tool:<8}  {stamp}  {session.title}")
            lines.append(f"            {resume_line(session)}")
        blocks.append("\n".join(lines))
    return "\n\n".join(blocks) + "\n"
```

```python
def main(
    argv: list[str] | None = None,
    *,
    now: datetime | None = None,
    home: Path | None = None,
) -> int:
    args = list(sys.argv[1:] if argv is None else argv)
    since = "7d"
    mode = ""
    index = 0
    while index < len(args):
        arg = args[index]
        if arg == "--since":
            if index + 1 >= len(args):
                print(USAGE, file=sys.stderr)
                return 1
            since = args[index + 1]
            index += 2
            continue
        if arg == "tui" and mode == "":

            index += 1
            continue

        return 1

        return 1

        print(warning, file=sys.stderr)
    if failures == 3:

    if mode == "tui":
        print(USAGE, file=sys.stderr)
        return 1
    sys.stdout.write(render(groups, now=current, since_label=since))
    return 0
```
